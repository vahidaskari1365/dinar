    59→/* ---------- helpers ---------- */
    60→function parseNum(s: string): number {
    61→  const fa = "۰۱۲۳۴۵۶۷۸۹";
    62→  const ar = "٠١٢٣٤٥٦٧٨٩";
    63→  let out = "";
    64→  for (const ch of s) {
    65→    const fi = fa.indexOf(ch);
    66→    const ai = ar.indexOf(ch);
    67→    out += fi >= 0 ? String(fi) : ai >= 0 ? String(ai) : ch;
    68→  }
    69→  return Number(out.replace(/[,\s٬]/g, "")) || 0;
    70→}

    82→function fmtAsset(asset: string, amount: number): string {
    83→  if (asset === "IRR") return `${toFa(amount)} تومان`;
    84→  if (asset === "USDT") return `${toFa(amount, 2)} تتر`;
    85→  if (asset === "GOLD") return `${toFa(amount, 3)} گرم`;
    86→  return toFa(amount);
    87→}

    89→/* ---------- hook ---------- */
    90→function useWallet() {
    91→  const [data, setData] = useState<WalletData | null>(null);
    92→  const [loading, setLoading] = useState(true);
    93→  const refresh = useCallback(async () => {
    94→    try {
    95→      const res = await fetch("/api/wallet", { cache: "no-store" });
    96→      if (res.ok) setData(await res.json());
    97→    } finally {
    98→      setLoading(false);
    99→    }
   100→  }, []);
   101→  useEffect(() => {
   102→    refresh();
   103→  }, [refresh]);
   104→  return { data, loading, refresh };
   105→}

   107→function usePrices(intervalMs = 4000) {
   108→  const [prices, setPrices] = useState<Prices | null>(null);
   109→  const histRef = useRef<number[]>([]);
   110→  const [hist, setHist] = useState<number[]>([]);
   111→
   112→  useEffect(() => {
   113→    let alive = true;
   114→    const tick = () => {
   115→      const p = currentPrices();
   116→      if (!alive) return;
   117→      setPrices(p);
   118→      histRef.current = [...histRef.current.slice(-47), p.gold];
   119→      setHist(histRef.current);
   120→    };
   121→    tick();
   122→    const id = setInterval(tick, intervalMs);
   123→    return () => {
   124→      alive = false;
   125→      clearInterval(id);
   126→    };
   127→  }, [intervalMs]);
   128→
   129→  return { prices, hist };
   130→}

   751→function ConvertForm({
   752→  busy,
   753→  prices,
   754→  wallet,
   755→  onConvert,
   756→}: {
   757→  busy: string | null;
   758→  prices: Prices;
   759→  wallet: { irr: number; usdt: number; gold: number };
   760→  onConvert: (from: "IRR" | "USDT" | "GOLD", to: "IRR" | "USDT" | "GOLD", amount: number) => void;
   761→}) {
   762→  const ASSETS = [
   763→    { k: "IRR" as const, l: "تومان", max: wallet.irr, dec: 0 },
   764→    { k: "USDT" as const, l: "تتر", max: wallet.usdt, dec: 2 },
   765→    { k: "GOLD" as const, l: "طلا (گرم)", max: wallet.gold, dec: 3 },
   766→  ];
   767→  const [from, setFrom] = useState<"IRR" | "USDT" | "GOLD">("IRR");
   768→  const [to, setTo] = useState<"IRR" | "USDT" | "GOLD">("USDT");
   769→  const [amount, setAmount] = useState("");
   770→
   771→  const fromA = ASSETS.find((a) => a.k === from)!;
   772→  const amt = parseNum(amount);
   773→  const result = useMemo(() => (amt > 0 ? convert(amt, from, to, prices) : 0), [amt, from, to, prices]);
   774→  const valid = amt > 0 && amt <= fromA.max && from !== to;
   775→
   776→  return (
   777→    <>
   778→      <DialogHeader className="text-right">
   779→        <DialogTitle className="font-black tracking-tight">تبدیل دارایی</DialogTitle>
   780→        <DialogDescription>نرخ لحظه‌ای با اسپرد شفاف ۰٫۶٪.</DialogDescription>
   781→      </DialogHeader>
   782→
   783→      <div className="grid gap-3">
   784→          <div className="rounded-2xl border border-border bg-background/40 p-4">
   785→            <div className="flex items-center justify-between gap-3">
   786→              <Label className="shrink-0">از</Label>
   787→              <div className="flex flex-1 items-center gap-2">
   788→                <Input
   789→                  dir="ltr"
   790→                  inputMode="decimal"
   791→                  className="h-10 flex-1 rounded-lg bg-background/60 text-left font-mono tnum"
   792→                  value={amount}
   793→                  onChange={(e) => setAmount(e.target.value)}
   794→                  placeholder="0"
   795→                />
   796→                <select
   797→                  value={from}
   798→                  onChange={(e) => setFrom(e.target.value as typeof from)}
   799→                  aria-label="از دارایی"
   800→                  className="h-10 rounded-lg border border-border bg-background/80 px-2 text-xs"
   801→                >
   802→                  {ASSETS.map((a) => (
   803→                    <option key={a.k} value={a.k}>
   804→                      {a.l}
   805→                    </option>
   806→                  ))}
   807→                </select>
   808→              </div>
   809→            </div>
   810→            <p className="tnum mt-2 text-left font-mono text-[10px] text-muted-foreground" dir="ltr">
   811→              حداکثر: {toFa(fromA.max, fromA.dec)}
   812→            </p>
   813→          </div>
   814→
   815→          <button
   816→            onClick={() => {
   817→              const f = from;
   818→              setFrom(to);
   819→              setTo(f);
   820→            }}
   821→            aria-label="جابجایی جهت تبدیل"
   822→            className="mx-auto grid size-9 place-items-center rounded-full border border-primary/30 bg-primary/10 text-primary transition-transform hover:rotate-180"
   823→          >
   824→            <ArrowLeftRight className="size-4" aria-hidden />
   825→          </button>
   826→
   827→          <div className="rounded-2xl border border-primary/25 bg-primary/[0.05] p-4">
   828→            <div className="flex items-center justify-between gap-3">
   829→              <Label className="shrink-0 text-primary">به</Label>
   830→              <div className="flex flex-1 items-center justify-end gap-2">
   831→                <p className="tnum flex-1 text-left font-mono text-lg font-black" dir="ltr">
   832→                  {result > 0 ? toFa(result, to === "IRR" ? 0 : to === "GOLD" ? 3 : 2) : "۰"}
   833→                </p>
   834→                <select
   835→                  value={to}
   836→                  onChange={(e) => setTo(e.target.value as typeof to)}
   837→                  aria-label="به دارایی"
   838→                  className="h-10 rounded-lg border border-border bg-background/80 px-2 text-xs"

   845→                </select>
   846→              </div>
   847→            </div>

   850→          <Button
   851→            disabled={!valid || !!busy}
   852→            onClick={() => onConvert(from, to, amt)}
   853→            className="h-12 rounded-2xl font-bold active:scale-[0.98]"

   855→            {busy ? <Loader2 className="size-4 animate-spin" aria-hidden /> : "انجام تبدیل"}
