    61→/* ---------- helpers ---------- */
    62→function parseNum(s: string): number {
    63→  const fa = "۰۱۲۳۴۵۶۷۸۹";
    64→  const ar = "٠١٢٣٤٥٦٧٨٩";
    65→  let out = "";
    66→  for (const ch of s) {
    67→    const fi = fa.indexOf(ch);
    68→    const ai = ar.indexOf(ch);
    69→    out += fi >= 0 ? String(fi) : ai >= 0 ? String(ai) : ch;
    70→  }
    71→  return Number(out.replace(/[,\s٬]/g, "")) || 0;
    72→}

    84→function fmtAsset(asset: string, amount: number): string {
    85→  if (asset === "IRR") return `${toFa(amount)} تومان`;
    86→  if (asset === "USDT") return `${toFa(amount, 2)} تتر`;
    87→  if (asset === "GOLD") return `${toFa(amount, 3)} گرم`;
    88→  return toFa(amount);
    89→}

    91→/* ---------- hook ---------- */
    92→function useWallet() {
    93→  const [data, setData] = useState<WalletData | null>(null);
    94→  const [loading, setLoading] = useState(true);
    95→  const refresh = useCallback(async () => {
    96→    try {
    97→      const res = await fetch("/api/wallet", { cache: "no-store" });
    98→      if (res.ok) setData(await res.json());
    99→    } finally {
   100→      setLoading(false);
   101→    }
   102→  }, []);
   103→  useEffect(() => {
   104→    refresh();
   105→  }, [refresh]);
   106→  return { data, loading, refresh };
   107→}

   109→function usePrices(intervalMs = 4000) {
   110→  const [prices, setPrices] = useState<Prices | null>(null);
   111→  const histRef = useRef<number[]>([]);
   112→  const [hist, setHist] = useState<number[]>([]);
   113→
   114→  useEffect(() => {
   115→    let alive = true;
   116→    const tick = () => {
   117→      const p = currentPrices();
   118→      if (!alive) return;
   119→      setPrices(p);
   120→      histRef.current = [...histRef.current.slice(-47), p.gold];
   121→      setHist(histRef.current);
   122→    };
   123→    tick();
   124→    const id = setInterval(tick, intervalMs);
   125→    return () => {
   126→      alive = false;
   127→      clearInterval(id);
   128→    };
   129→  }, [intervalMs]);
   130→
   131→  return { prices, hist };
   132→}

   757→function ConvertForm({
   758→  busy,
   759→  prices,
   760→  wallet,
   761→  onConvert,
   762→}: {
   763→  busy: string | null;
   764→  prices: Prices;
   765→  wallet: { irr: number; usdt: number; gold: number };
   766→  onConvert: (from: "IRR" | "USDT" | "GOLD", to: "IRR" | "USDT" | "GOLD", amount: number) => void;
   767→}) {
   768→  const ASSETS = [
   769→    { k: "IRR" as const, l: "تومان", max: wallet.irr, dec: 0 },
   770→    { k: "USDT" as const, l: "تتر", max: wallet.usdt, dec: 2 },
   771→    { k: "GOLD" as const, l: "طلا (گرم)", max: wallet.gold, dec: 3 },
   772→  ];
   773→  const [from, setFrom] = useState<"IRR" | "USDT" | "GOLD">("IRR");
   774→  const [to, setTo] = useState<"IRR" | "USDT" | "GOLD">("USDT");
   775→  const [amount, setAmount] = useState("");
   776→
   777→  const fromA = ASSETS.find((a) => a.k === from)!;
   778→  const amt = parseNum(amount);
   779→  const result = useMemo(() => (amt > 0 ? convert(amt, from, to, prices) : 0), [amt, from, to, prices]);
   780→  const valid = amt > 0 && amt <= fromA.max && from !== to;
   781→
   782→  return (
   783→    <>
   784→      <DialogHeader className="text-right">
   785→        <DialogTitle className="font-black tracking-tight">تبدیل دارایی</DialogTitle>

   795→                  dir="ltr"
   796→                  inputMode="decimal"
   797→                  className="h-10 flex-1 rounded-lg bg-background/60 text-left font-mono tnum"
   798→                  value={amount}
   799→                  onChange={(e) => setAmount(e.target.value)}
   800→                  placeholder="0"
   801→                />
   802→                <select
   803→                  value={from}
   804→                  onChange={(e) => setFrom(e.target.value as typeof from)}
   805→                  aria-label="از دارایی"
   806→                  className="h-10 rounded-lg border border-border bg-background/80 px-2 text-xs"
   807→                >
   808→                  {ASSETS.map((a) => (

   811→                    </option>
   812→                  ))}
   813→                </select>
   814→              </div>
   815→            </div>
   816→            <p className="tnum mt-2 text-left font-mono text-[10px] text-muted-foreground" dir="ltr">
