    62→    62→function parseNum(s: string): number {
    63→    63→  const fa = "۰۱۲۳۴۵۶۷۸۹";
    64→    64→  const ar = "٠١٢٣٤٥٦٧٨٩";
    65→    65→  let out = "";
    66→    66→  for (const ch of s) {
    67→    67→    const fi = fa.indexOf(ch);
    68→    68→    const ai = ar.indexOf(ch);
    69→    69→    out += fi >= 0 ? String(fi) : ai >= 0 ? String(ai) : ch;
    70→    70→  }
    71→    71→  return Number(out.replace(/[,\s٬]/g, "")) || 0;
    72→    72→}

    84→    84→function fmtAsset(asset: string, amount: number): string {
    85→    85→  if (asset === "IRR") return `${toFa(amount)} تومان`;
    86→    86→  if (asset === "USDT") return `${toFa(amount, 2)} تتر`;
    87→    87→  if (asset === "GOLD") return `${toFa(amount, 3)} گرم`;
    88→    88→  return toFa(amount);
    89→    89→}

    92→    92→function useWallet() {
    93→    93→  const [data, setData] = useState<WalletData | null>(null);
    94→    94→  const [loading, setLoading] = useState(true);
    95→    95→  const refresh = useCallback(async () => {

   100→   100→      setLoading(false);

   103→   103→  useEffect(() => {

   109→   109→function usePrices(intervalMs = 4000) {
   110→   110→  const [prices, setPrices] = useState<Prices | null>(null);
   111→   111→  const histRef = useRef<number[]>([]);
   112→   112→  const [hist, setHist] = useState<number[]>([]);
   113→   113→
   114→   114→  useEffect(() => {
   115→   115→    let alive = true;
   116→   116→    const tick = () => {
   117→   117→      const p = currentPrices();
   118→   118→      if (!alive) return;
   119→   119→      setPrices(p);
   120→   120→      histRef.current = [...histRef.current.slice(-47), p.gold];
   121→   121→      setHist(histRef.current);
   122→   122→    };
   123→   123→    tick();
   124→   124→    const id = setInterval(tick, intervalMs);
   125→   125→    return () => {
   126→   126→      alive = false;
   127→   127→      clearInterval(id);
   128→   128→    };
   129→   129→  }, [intervalMs]);
   130→   130→
   131→   131→  return { prices, hist };
   132→   132→}
